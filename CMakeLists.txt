cmake_minimum_required(VERSION 3.20)
project(CppMusicPlayer VERSION 2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.32.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL2)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.92.2b GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC SDL2::SDL2)
FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git GIT_TAG 0.11.21 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(miniaudio)
FetchContent_Declare(sqlite URL https://www.sqlite.org/2025/sqlite-amalgamation-3530400.zip)
FetchContent_MakeAvailable(sqlite)
add_library(sqlite3 STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3 PUBLIC ${sqlite_SOURCE_DIR})
FetchContent_Declare(utf8cpp GIT_REPOSITORY https://github.com/nemtrif/utfcpp.git GIT_TAG v4.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(utf8cpp)
set(utf8cpp_INCLUDE_DIR "${utf8cpp_SOURCE_DIR}/source" CACHE PATH "utf8cpp headers")
FetchContent_Declare(taglib GIT_REPOSITORY https://github.com/taglib/taglib.git GIT_TAG v2.1.1 GIT_SHALLOW TRUE)
set(TAGLIB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TAGLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TAGLIB_BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(taglib)
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(stb)
enable_testing()
add_executable(AuroraCoreTests tests/core_tests.cpp src/Queue.cpp src/Search.cpp)
target_include_directories(AuroraCoreTests PRIVATE include)
add_test(NAME AuroraCoreTests COMMAND AuroraCoreTests)
if(MSVC)
 target_compile_options(AuroraCoreTests PRIVATE /W4 /permissive-)
else()
 target_compile_options(AuroraCoreTests PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(CppMusicPlayer src/main.cpp src/App.cpp src/AudioPlayer.cpp src/Database.cpp src/MusicLibrary.cpp src/Search.cpp src/Queue.cpp src/Metadata.cpp src/Settings.cpp src/UI.cpp)
target_include_directories(CppMusicPlayer PRIVATE include ${miniaudio_SOURCE_DIR} ${stb_SOURCE_DIR})
target_link_libraries(CppMusicPlayer PRIVATE imgui SDL2::SDL2 sqlite3 TagLib::tag)
if(WIN32)
 target_link_libraries(CppMusicPlayer PRIVATE shell32 ole32 comdlg32)
 target_compile_definitions(CppMusicPlayer PRIVATE NOMINMAX UNICODE _UNICODE)
endif()
if(MSVC)
 target_compile_options(CppMusicPlayer PRIVATE /W4 /permissive-)
else()
 target_compile_options(CppMusicPlayer PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET CppMusicPlayer POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:CppMusicPlayer>/data")
message(STATUS "Aurora Player 2.0 configured. Build with: cmake --build build --config Release")
